    tensor_cls,

    storage_size_bytes,

    requires_grad,
    ref_counter_handle,
    ref_counter_offset,
    event_handle,
    event_sync_required,
):

        storage = storage_cls(0, dtype=dtype, device=storage_device, _internal=True)

            storage_cls, (storage_handle, storage_offset_bytes)
        )

                storage_device,
                storage_handle,
                storage_size_bytes,
                storage_offset_bytes,
                ref_counter_handle,
                ref_counter_offset,
                event_handle,
                event_sync_required,
            )

                storage
